function waLink(wa, nombre, equipo) {
  if (!wa) return null;
  var digits = String(wa).replace(/\D/g, '');
  if (digits.charAt(0) === '0') digits = digits.slice(1);
  if (digits.slice(0, 2) !== '54') digits = '54' + digits;
  var msg = encodeURIComponent('Hola ' + nombre + ', tu equipo ' + equipo + ' está listo para retirar en Tecnopuerto. 📱');
  return 'https://wa.me/' + digits + '?text=' + msg;
}

  if (digits.slice(0, 2) !== '54') digits = '54' + digits;

}
function _waDigits(wa) {
  var digits = String(wa || '').replace(/\D/g, '');
  if (!digits) return '';
  if (digits.charAt(0) === '0') digits = digits.slice(1);
  if (digits.slice(0, 2) !== '54') digits = '54' + digits;
  return digits;
}
// Link de WhatsApp simple (sin mensaje precargado) — se usa en la pestaña Clientes
function waLinkPlano(wa) {

// CLIENTES: se arma solo a partir de Reparaciones (sin hoja propia, sin campo obligatorio).
// Agrupa por WhatsApp si está cargado; si no, por nombre (mayúsculas, sin espacios extra).
function _clientesAgregados() {
  var map = {};
  state.reparaciones.forEach(function(r) {
    var wa = _waDigits(r.whatsapp);
    var nombre = String(r.cliente || '').trim();
    if (!wa && !nombre) return; // sin nombre ni whatsapp, no hay con qué agrupar
    var key = wa || ('n:' + nombre.toUpperCase());
    if (!map[key]) map[key] = { nombre: nombre || '(sin nombre)', whatsapp: r.whatsapp || '', cantidad: 0, total: 0, ultima: '' };
    map[key].cantidad++;
    map[key].total += parseFloat(r.precioReparacion) || 0;
    if (nombre) map[key].nombre = nombre;
    if (!map[key].whatsapp && r.whatsapp) map[key].whatsapp = r.whatsapp;
    var f = r.fechaEntrega || r.fecha || '';
    if (f && f > map[key].ultima) map[key].ultima = f;
  });
  return Object.keys(map).map(function(k) { return map[k]; });
}
function renClientes() {
  if (!el('tClientes')) return;
  var buscar = (el('bClientes') ? (el('bClientes').value || '') : '').toLowerCase();
  var arr = _clientesAgregados().filter(function(c) {
    return !buscar || c.nombre.toLowerCase().indexOf(buscar) >= 0 || String(c.whatsapp).indexOf(buscar) >= 0;
  }).sort(function(a, b) { return b.cantidad - a.cantidad || String(b.ultima).localeCompare(String(a.ultima)); });
  if (el('clientesRes')) el('clientesRes').textContent = arr.length + (arr.length === 1 ? ' cliente' : ' clientes');
  if (!arr.length) { el('tClientes').innerHTML = '<div class="empty">Todavía no hay clientes cargados (se arman solos desde Reparaciones)</div>'; return; }
  el('tClientes').innerHTML = '<table><thead><tr><th>Cliente</th><th>WhatsApp</th><th>Reparaciones</th><th>Total facturado</th><th>Última visita</th></tr></thead><tbody>' +
    arr.map(function(c) {
      var wl = waLinkPlano(c.whatsapp);
      return '<tr><td>' + c.nombre + '</td>' +
        '<td>' + (wl ? '<a href="' + wl + '" target="_blank" style="color:#25D366;text-decoration:none">💬 ' + c.whatsapp + '</a>' : (c.whatsapp || '—')) + '</td>' +
        '<td class="mono">' + c.cantidad + '</td>' +
        '<td class="mono ts">' + fm(c.total) + '</td>' +
        '<td class="mono tm">' + fd(c.ultima) + '</td></tr>';
    }).join('') + '</tbody></table>';
  // Datalist para autocompletar el campo Cliente en "Nueva reparación"
  if (el('clienteList')) {
    el('clienteList').innerHTML = arr.map(function(c) { return '<option value="' + c.nombre.replace(/"/g, '&quot;') + '">'; }).join('');
  }
}

// Fecha LOCAL en formato YYYY-MM-DD — toISOString() usa UTC y desfasaba el día
// (ej: una venta a las 21:30 en Argentina -UTC-3- ya caía en el día siguiente en UTC).
function localDateStr(d) {
  d = d || new Date();
  var tzMs = d.getTimezoneOffset() * 60000;
  return new Date(d.getTime() - tzMs).toISOString().split('T')[0];
}
function today() { return localDateStr(); }
function nowTime() { var d = new Date(); return ('0' + d.getHours()).slice(-2) + ':' + ('0' + d.getMinutes()).slice(-2); }
// Clave de orden: fecha + hora (si no hay hora, usa el id que es cronologico)
function ventaKey(v) { return (v.fecha || '') + 'T' + (v.hora || '') + '#' + (v.id || ''); }
// Partes del pago de una venta: [{medio, monto}] — soporta pago combinado (medio1+medio2)
function medioPartes(v) {
  var total = (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0);
  if (v.medio2) {
    var m2 = parseFloat(v.montoMedio2) || 0;
    var m1 = v.montoMedio1 !== undefined && v.montoMedio1 !== '' ? (parseFloat(v.montoMedio1) || 0) : (total - m2);
    return [{ medio: v.medio1 || v.medio, monto: m1 }, { medio: v.medio2, monto: m2 }];
  }
  // Venta con cobro parcial: a la caja del día DE LA VENTA solo entra lo efectivamente cobrado ese
  // día (montoMedio1 guarda "lo cobrado ahora", no el total). El saldo entra a la caja del día en
  // que se cobra — ver calcCajaDia().
  if ((parseFloat(v.saldoPendiente) || 0) > 0) {
    return [{ medio: v.medio, monto: parseFloat(v.montoMedio1) || 0 }];
  }
  return [{ medio: v.medio, monto: total }];
}
// Partes del pago de una reparación entregada: [{medio, monto}] — mismo patrón que medioPartes(v),
// el medio recién se define al marcar Entregado (ver confirmarEntrega/abrirModalEntrega).
function repMedioPartes(r) {
  var total = parseFloat(r.precioReparacion) || 0;
  if (r.medio2) {
    var m2 = parseFloat(r.montoMedio2) || 0;
    var m1 = r.montoMedio1 !== undefined && r.montoMedio1 !== '' ? (parseFloat(r.montoMedio1) || 0) : (total - m2);
    return [{ medio: r.medio1 || r.medio, monto: m1 }, { medio: r.medio2, monto: m2 }];
  }
  return [{ medio: r.medio || 'Otro', monto: total }];
}
// Etiqueta corta para mostrar el medio de una reparación en tablas ("Efectivo" o "Efectivo + Transferencia")
function _medioLabelRep(r) {
  if (!r.medio && !r.medio2) return '—';
  return r.medio2 ? (r.medio1 || r.medio) + ' + ' + r.medio2 : r.medio;
}

function apiGet(hoja, cb) {
  var url = WURL + '?accion=leer&hoja=' + hoja;
  fetch(url).then(function(r) { return r.json(); }).then(function(d) { cb(null, d.data); }).catch(function(e) { cb(e); });
}
function apiPost(body, cb) {
  fetch(WURL, { method: 'POST', body: JSON.stringify(body) })
    .then(function(r) { return r.json(); })
    .then(function(d) { cb(null, d); })
    .catch(function(e) { cb(e); });
}

function cargarTodo() {
  el('spin').style.display = 'inline-block';
  var loaded = 0;
  var errs = 0;
  function done(err, hoja, data) {
    if (err) { errs++; } else { state[hoja] = data; }
    loaded++;
    if (loaded === 5) {
      el('spin').style.display = 'none';
      if (errs === 0) { setConn(true); renAll(); } else { setConn(false); }
    }
  }
  apiGet('stock', function(e, d) { done(e, 'stock', d); });
  apiGet('ventas', function(e, d) { done(e, 'ventas', d); });
  apiGet('reparaciones', function(e, d) { done(e, 'reparaciones', d); });
  apiGet('gastos', function(e, d) { done(e, 'gastos', d); });
  apiGet('cierres', function(e, d) { done(e, 'cierres', d); });
}

// ganancia neta en Reparaciones por mes. Cuenta reparaciones ENTREGADAS este mes (mismo criterio
// que "Ganancia hoy" y Estadísticas: fechaEntrega, con fallback a fecha) — ganancia = precio - costo técnico.
var META_REP_GANANCIA = 800000;
function renMeta() {
  if (!el('metaCard')) return;
  var mesActual = localDateStr(new Date()).slice(0, 7); // 'YYYY-MM'
  var ent = state.reparaciones.filter(function(r) { return r.estado === 'Entregado' && String(r.fechaEntrega || r.fecha || '').slice(0, 7) === mesActual; });
  var gan = ent.reduce(function(s, r) { return s + (parseFloat(r.precioReparacion) || 0) - (parseFloat(r.costoRepuesto) || 0); }, 0);
  var pct = Math.min(100, Math.round(gan / META_REP_GANANCIA * 100));
  if (el('metaTxt')) el('metaTxt').textContent = fm(gan) + ' / ' + fm(META_REP_GANANCIA);
  if (el('metaBar')) {
    el('metaBar').style.width = pct + '%';
    el('metaBar').style.background = gan >= META_REP_GANANCIA ? 'var(--success)' : 'var(--accent)';
  }
  if (el('metaSub')) {
    el('metaSub').textContent = gan >= META_REP_GANANCIA
      ? '🎉 Meta cumplida este mes'
      : fm(META_REP_GANANCIA - gan) + ' más para llegar a la meta';
  }
}

function _statsStockValor(s, campo) {
  var c = parseInt(s.cantidad) || 0;
  var costo = parseFloat(s.costo) || 0;
  var pv = parseFloat(s.precioVenta) || 0;
  switch (campo) {
    case 'nombre': return String(s.nombre || '').toLowerCase();
    case 'costo': return costo;
    case 'precioVenta': return pv;
    case 'margen': return costo > 0 ? (pv - costo) / costo * 100 : -Infinity;
    case 'valor': return costo * c;
    default: return c; // cantidad
  }
}
function renStatsStock() {
  var buscar = (el('bStatsStock') ? (el('bStatsStock').value || '') : '').toLowerCase();
  var campo = el('statsStockOrden') ? el('statsStockOrden').value : 'cantidad';
  var fil = state.stock.filter(function(s) { return !buscar || s.nombre.toLowerCase().indexOf(buscar) >= 0; });
  var totC = state.stock.reduce(function(s, i) { return s + (parseFloat(i.costo) || 0) * (parseInt(i.cantidad) || 0); }, 0);
  var sin = state.stock.filter(function(s) { return (parseInt(s.cantidad) || 0) === 0; }).length;
  var bajo = state.stock.filter(function(s) { var c = parseInt(s.cantidad) || 0; var m = parseInt(s.stockMinimo) || 3; return c > 0 && c <= m; }).length;
  if (el('statsStockRes')) el('statsStockRes').innerHTML = state.stock.length + ' productos · <span class="mono tw">' + fm(totC) + '</span>' + (sin ? ' · <span class="td">' + sin + ' sin stock</span>' : '') + (bajo ? ' · <span class="tw">' + bajo + ' bajo mínimo</span>' : '');
  if (!el('tStatsStock')) return;
  if (!fil.length) { el('tStatsStock').innerHTML = '<div class="empty">Sin resultados</div>'; return; }
  var sorted = fil.slice().sort(function(a, b) {
    var va = _statsStockValor(a, campo), vb = _statsStockValor(b, campo);
    if (va < vb) return -1 * statsStockDir;
    if (va > vb) return 1 * statsStockDir;
    return 0;
  });
  el('tStatsStock').innerHTML = '<table><thead><tr><th>Producto</th><th>Stock</th><th>Mín.</th><th>Costo unit.</th><th>Precio venta</th><th>% Ganancia</th><th>Valor total</th><th>Estado</th></tr></thead><tbody>' +
    sorted.map(function(s) {
      var c = parseInt(s.cantidad) || 0;
      var mn = parseInt(s.stockMinimo) || 3;
      var costo = parseFloat(s.costo) || 0;
      var pv = parseFloat(s.precioVenta) || 0;
      var marg = costo > 0 && pv > 0 ? (pv - costo) / costo * 100 : null;
      var cls = c === 0 ? 'tgd' : c <= mn ? 'tgw' : 'tgs';
      var lbl = c === 0 ? 'Sin stock' : c <= mn ? 'Bajo' : 'OK';
      return '<tr><td>' + s.nombre + '</td><td class="mono">' + c + '</td><td class="mono tm">' + mn + '</td><td class="mono">' + fm(costo) + '</td><td class="mono">' + (pv > 0 ? fm(pv) : '—') + '</td><td class="mono ' + (marg === null ? 'tm' : marg >= 80 ? 'ts' : 'tw') + '">' + (marg === null ? '—' : marg.toFixed(0) + '%') + '</td><td class="mono">' + fm(costo * c) + '</td><td><span class="tag ' + cls + '">' + lbl + '</span></td></tr>';
    }).join('') + '</tbody></table>';
}

function renDash() {
  var hoy = today();
  var vH = state.ventas.filter(function(v) { return v.fecha === hoy; });
  var hace7 = new Date(); hace7.setDate(hace7.getDate() - 7);
  var vS = state.ventas.filter(function(v) { return new Date(v.fecha + 'T12:00:00') >= hace7; });
  function sv(arr) { return arr.reduce(function(s, v) { return s + (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0); }, 0); }
  function sc(arr) { return arr.reduce(function(s, v) { return s + (parseFloat(v.costoUnit) || 0) * (parseInt(v.cantidad) || 0); }, 0); }
  var tH = sv(vH), cH = sc(vH), mH = cH > 0 ? ((tH - cH) / cH * 100) : 0;
  var tS = sv(vS), cS = sc(vS), mS = cS > 0 ? ((tS - cS) / cS * 100) : 0;
  var rP = state.reparaciones.filter(function(r) { return r.estado === 'En proceso'; });
  // Ganancia neta de Reparaciones ENTREGADAS hoy (mismo criterio que Estadísticas: fechaEntrega,
  // con fallback a fecha) — se suma a la ganancia de artículos para dar el número combinado del día.
  var rEntHoy = state.reparaciones.filter(function(r) { return r.estado === 'Entregado' && (r.fechaEntrega || r.fecha) === hoy; });
  var ganRepHoy = rEntHoy.reduce(function(s, r) { return s + (parseFloat(r.precioReparacion) || 0) - (parseFloat(r.costoRepuesto) || 0); }, 0);
  var ganArtHoy = tH - cH;
  // "Vendido hoy" es la venta total del día (lo que se factura, no lo que entró en efectivo/transferencia).
  // Si hoy se generaron ventas con cobro parcial todavía sin cobrar, se aclara acá para que no se
  // confunda con lo que realmente entró a la Caja de hoy (esa sí solo suma lo efectivamente cobrado).
  var saldoPendHoy = vH.reduce(function(s, v) { return s + ((parseFloat(v.saldoPendiente) || 0) > 0 && !v.cobradoSaldo ? parseFloat(v.saldoPendiente) : 0); }, 0);
  el('kH').textContent = fm(tH);
  el('kHs').textContent = vH.length + ' operación' + (vH.length !== 1 ? 'es' : '') + ' · margen ' + mH.toFixed(0) + '%' + (saldoPendHoy > 0 ? ' · ' + fm(saldoPendHoy) + ' aún sin cobrar' : '');
  el('kGv').textContent = fm(ganArtHoy + ganRepHoy);
  el('kG').textContent = 'Artículos: ' + fm(ganArtHoy) + ' · Reparaciones: ' + fm(ganRepHoy);
  el('kS').textContent = fm(tS);
  el('kSs').textContent = vS.length + ' venta' + (vS.length !== 1 ? 's' : '');
  el('kM').textContent = mS.toFixed(0) + '%';
  var ticketProm = vS.length ? tS / vS.length : 0;
  if (el('kTicket')) el('kTicket').textContent = fm(ticketProm);

  // Necesita tu atención: en proceso / listas para retirar / stock bajo (mismo criterio que Stock:
  // cantidad > 0 y <= stockMinimo, que ya es editable por producto en esa solapa) / caja de hoy sin cerrar
  var rListo = state.reparaciones.filter(function(r) { return r.estado === 'Listo'; });
  var bajoStock = state.stock.filter(function(s) { var c = parseInt(s.cantidad) || 0; var m = parseInt(s.stockMinimo) || 3; return c > 0 && c <= m; });
  if (el('kProc')) el('kProc').textContent = rP.length;
  if (el('kListo')) el('kListo').textContent = rListo.length;
  if (el('kStockBajo')) el('kStockBajo').textContent = bajoStock.length;
  // Hovercards: al pasar el mouse, mostrar el detalle sin tener que ir a la pestaña correspondiente.
  function _hcRep(arr) {
    if (!arr.length) return '<div class="tm">Sin pendientes</div>';
    return arr.slice(0, 10).map(function(r) { return '<div class="hcrow"><span>' + r.cliente + '</span><span class="tm">' + r.equipo + '</span></div>'; }).join('');
  }
  if (el('popProc')) el('popProc').innerHTML = '<div class="hctitle">En proceso</div>' + _hcRep(rP);
  if (el('popListo')) el('popListo').innerHTML = '<div class="hctitle">Listas para retirar</div>' + _hcRep(rListo);
  if (el('popStockBajo')) {
    var top10 = bajoStock.slice().sort(function(a, b) { return (parseInt(a.cantidad) || 0) - (parseInt(b.cantidad) || 0); }).slice(0, 10);
    el('popStockBajo').innerHTML = '<div class="hctitle">Stock bajo (primeros 10)</div>' + (top10.length
      ? top10.map(function(s) { return '<div class="hcrow"><span>' + s.nombre + '</span><span class="tw mono">' + (parseInt(s.cantidad) || 0) + '</span></div>'; }).join('')
      : '<div class="tm">Sin artículos bajo mínimo</div>');
  }

  var cd = calcCajaDia(hoy);
  window._cajaMediosHoy = cd.medios; window._cajaTotHoy = cd.total;
  var cierreHoyExiste = state.cierres.some(function(c) { return c.fecha === hoy; });
  if (el('kCajaHoy')) {

  }
  if (el('popCajaHoy')) {
    var mediosHoy = Object.entries(cd.medios).sort(function(a, b) { return b[1] - a[1]; });
    el('popCajaHoy').innerHTML = '<div class="hctitle">Caja de hoy por medio</div>' + (mediosHoy.length
      ? mediosHoy.map(function(x) { return '<div class="hcrow"><span>' + x[0] + '</span><span class="mono ts">' + fm(x[1]) + '</span></div>'; }).join('')

  if (el('cajaDia')) {

        Object.entries(cd.medios).sort(function(a,b){return b[1]-a[1];}).map(function(x){return '<div class="irow" style="min-width:150px"><span class="tm">'+x[0]+'</span><span class="mono ts">'+fm(x[1])+'</span></div>';}).join('') +

    '</tbody></table>';

      }).join('') + '</tbody></table>';
  }

function _medioToCol(m) {
  switch (m) {
    case 'Efectivo': return 'efectivo';
    case 'Transferencia': return 'transferencia';
    case 'Débito': return 'debito';
    case 'Crédito': return 'credito';
    case 'Pago QR': return 'pagoQR';
    default: return 'otros';
  }
}
// Calcula el desglose por medio de pago de UN día puntual (ventas + reparaciones entregadas ese día)
function calcCajaDia(fechaStr) {
  var medios = {};
  state.ventas.filter(function(v) { return v.fecha === fechaStr; })
    .forEach(function(v) { medioPartes(v).forEach(function(p) { var med = p.medio || 'Otro'; medios[med] = (medios[med] || 0) + p.monto; }); });
  state.reparaciones.filter(function(r) { return (r.fechaEntrega || r.fecha) === fechaStr && r.estado === 'Entregado'; })
    .forEach(function(r) { repMedioPartes(r).forEach(function(p) { var med = p.medio || 'Otro'; medios[med] = (medios[med] || 0) + p.monto; }); });
  // Saldos de ventas con cobro parcial que se terminan de cobrar este día (no el día de la venta original)
  state.ventas.filter(function(v) { return v.fechaCobroSaldo === fechaStr && v.cobradoSaldo; })
    .forEach(function(v) { var med = v.medioSaldo || v.medio || 'Otro'; medios[med] = (medios[med] || 0) + (parseFloat(v.saldoPendiente) || 0); });
  var total = Object.values(medios).reduce(function(s, v) { return s + v; }, 0);
  return { medios: medios, total: total };
}

function cerrarCaja(fechaOpt) {
  var fecha = fechaOpt || today();
  if (!fecha) { alert('Elegí una fecha.'); return; }
  if (state.cierres.some(function(c) { return c.fecha === fecha; })) { alert('Ese día ya tiene un cierre. Reabrilo primero si querés rehacerlo.'); return; }
  var cd = fecha === today() && window._cajaTotHoy !== undefined ? { medios: window._cajaMediosHoy, total: window._cajaTotHoy } : calcCajaDia(fecha);
  if (!cd.total) { alert('No hay ventas ni reparaciones entregadas ese día — nada para cerrar.'); return; }
  if (!confirm('¿Cerrar la caja del ' + fd(fecha) + ' por ' + fm(cd.total) + '?\nUna vez cerrada, este total queda fijo aunque después carguen algo con esa misma fecha.')) return;
  var row = { id: gid(), fecha: fecha, efectivo: 0, transferencia: 0, debito: 0, credito: 0, pagoQR: 0, otros: 0, total: cd.total, cerradoEn: nowTime() };
  Object.keys(cd.medios).forEach(function(m) { var c = _medioToCol(m); row[c] = (row[c] || 0) + cd.medios[m]; });
  apiPost({ accion: 'agregar', hoja: 'cierres', item: row }, function(err) {
    if (err) return alert('Error al cerrar caja.');
    state.cierres.push(row);
    renCierreCaja();
  });
}
function reabrirCierre(id) {
  if (!confirm('¿Reabrir este cierre? Se borra el registro congelado y volvés a ver los números en vivo de ese día (pueden haber cambiado si cargaron algo nuevo con esa fecha desde que se cerró).')) return;
  apiPost({ accion: 'borrar', hoja: 'cierres', id: id }, function(err) {
    if (err) return alert('Error al reabrir.');
    state.cierres = state.cierres.filter(function(c) { return c.id !== id; });
    renCierreCaja();
  });
}

// Nota: si hay pago combinado, el monto del 2do medio se SUMA al precio (no se descuenta) —
// "Precio de venta" pasa a ser solo la parte del 1er medio. Ver calcV() y regVenta().
function calcV() {
  var id = el('vProd').value;
  var prod = state.stock.filter(function(s) { return s.id === id; })[0];
  var precio = parseFloat(el('vPrice').value) || 0;
  var cant = parseInt(el('vCant').value) || 1;
  var combinado = el('vCombinado') && el('vCombinado').checked;
  var monto2 = combinado ? (parseFloat(el('vMonto2').value) || 0) : 0;
  if (el('vComboTotal')) el('vComboTotal').textContent = combinado ? 'Total de la venta: ' + fm(precio * cant + monto2) : '';
  if (!prod || precio <= 0) { el('iCalc').style.display = 'none'; return; }
  var total = precio * cant + monto2;
  var costoTot = (parseFloat(prod.costo) || 0) * cant;
  var gan = total - costoTot;
  var marg = costoTot > 0 ? (gan / costoTot * 100) : 0;
  el('iCalc').style.display = 'block';
  el('cGan').textContent = fm(gan);
  el('cMar').textContent = marg.toFixed(0) + '%';
  el('cMar').style.color = marg >= 80 ? 'var(--success)' : 'var(--warning)';
  // Alerta de margen bajo / venta por debajo del costo
  var costo = parseFloat(prod.costo) || 0;
  var w = el('cWarn');
  if (w) {
    if (total < costoTot) {
      w.style.display = 'block'; w.style.color = 'var(--danger)';
      w.textContent = '⚠ Estás vendiendo POR DEBAJO del costo (' + fm(costoTot) + ').';
    } else if (marg < 30) {
      w.style.display = 'block'; w.style.color = 'var(--warning)';
      w.textContent = '⚠ Margen bajo (menos del 30%). Costo: ' + fm(costoTot) + '.';
    } else {
      w.style.display = 'none'; w.textContent = '';
    }
  }
  if (el('vParcial') && el('vParcial').checked && el('vSaldoCalc')) {
    var montoAhoraPrev = parseFloat(el('vMontoAhora').value) || 0;
    el('vSaldoCalc').value = fm(total - montoAhoraPrev);
  }
}
function regVenta() {
  var id = el('vProd').value;
  var prod = window._prodSeleccionado;
  var cant = parseInt(el('vCant').value) || 0;
  var precio = parseFloat(el('vPrice').value) || 0;
  if (!prod || !precio || cant <= 0) return showMsg('msgV', 'd', 'Completá producto, cantidad y precio.');
  if (cant > (parseInt(prod.cantidad) || 0)) return showMsg('msgV', 'd', 'Stock insuficiente. Disponible: ' + prod.cantidad);
  if (!el('vMedio').value) return showMsg('msgV', 'd', 'Elegí el medio de pago.');
  var combinado = el('vCombinado') && el('vCombinado').checked;
  var parcial = el('vParcial') && el('vParcial').checked;
  var medio2 = '', montoMedio1 = precio * cant, montoMedio2 = 0, medioLabel = el('vMedio').value;
  if (combinado) {
    medio2 = el('vMedio2').value;
    montoMedio2 = parseFloat(el('vMonto2').value) || 0;
    if (!medio2) return showMsg('msgV', 'd', 'Elegí el segundo medio de pago.');
    if (medio2 === el('vMedio').value) return showMsg('msgV', 'd', 'Elegí dos medios distintos.');
    if (montoMedio2 <= 0) return showMsg('msgV', 'd', 'Ingresá el monto del segundo medio.');
    medioLabel = el('vMedio').value + ' + ' + medio2;
  }
  var total = montoMedio1 + montoMedio2; // total real de la venta (precio*cant, + monto2 si es combinado)
  var precioEfectivoUnit = total / cant; // precio por unidad ya incluyendo ambos medios — la Ganancia se calcula SIEMPRE sobre el total real, cobrado o no

  // Cobro parcial: se registra igual el total/margen real de la venta, pero a la Caja del día
  // solo entra lo que efectivamente se cobra hoy (ver medioPartes/calcCajaDia). El resto queda
  // como saldo pendiente con un recordatorio, visible en "Cobros pendientes".
  var clienteParcial = '', waParcial = '', montoAhora = 0, saldoPendiente = 0, fechaRecordatorio = '';
  if (parcial) {
    clienteParcial = (el('vParcialCliente').value || '').trim();
    waParcial = (el('vParcialWA').value || '').trim();
    montoAhora = parseFloat(el('vMontoAhora').value) || 0;
    fechaRecordatorio = el('vFechaRecordatorio').value || '';
    if (!clienteParcial) return showMsg('msgV', 'd', 'Ingresá el nombre del cliente para el cobro parcial.');
    if (montoAhora <= 0 || montoAhora >= total) return showMsg('msgV', 'd', 'El monto cobrado ahora tiene que ser mayor a $0 y menor al total de la venta (' + fm(total) + ').');
    if (!fechaRecordatorio) return showMsg('msgV', 'd', 'Elegí una fecha de recordatorio para cobrar el saldo.');
    saldoPendiente = total - montoAhora;
  }

  if (precioEfectivoUnit < (parseFloat(prod.costo) || 0) && !confirm('Estás vendiendo POR DEBAJO del costo (' + fm(prod.costo) + ').\n¿Registrar igual?')) return;
  var venta = {
    id: gid(), fecha: today(), hora: nowTime(), nombre: prod.nombre, cantidad: cant, precioVenta: precioEfectivoUnit, costoUnit: parseFloat(prod.costo) || 0,
    medio: medioLabel, medio1: el('vMedio').value, montoMedio1: parcial ? montoAhora : montoMedio1, medio2: medio2, montoMedio2: montoMedio2,
    cliente: parcial ? clienteParcial : '', clienteWA: parcial ? waParcial : '',
    saldoPendiente: saldoPendiente, fechaRecordatorio: fechaRecordatorio, cobradoSaldo: !parcial, fechaCobroSaldo: '', medioSaldo: ''
  };
  apiPost({ accion: 'agregar', hoja: 'ventas', item: venta }, function(err) {
    if (err) return showMsg('msgV', 'd', 'Error al guardar.');
    apiPost({ accion: 'actualizar', hoja: 'stock', id: prod.id, campo: 'cantidad', valor: (parseInt(prod.cantidad) || 0) - cant }, function() {});
    state.ventas.unshift(venta);
    prod.cantidad = (parseInt(prod.cantidad) || 0) - cant;
    renSelProd(); renVentas(); renStock(); renDash();
    el('vBusq').value = ''; el('vProd').value = ''; el('vPrice').value = ''; el('vCant').value = '1';
    el('vMedio').selectedIndex = 0;
    el('vCombinado').checked = false; el('vComboBox').style.display = 'none'; el('vMedio2').selectedIndex = 0; el('vMonto2').value = '';
    el('vParcial').checked = false; toggleParcial();
    if (el('lblVPrice')) el('lblVPrice').textContent = 'Precio de venta';
    el('iProf').style.display = 'none'; el('iCalc').style.display = 'none';
    showMsg('msgV', 's', '✓ Venta registrada — ' + fm(total) + (parcial ? ' (cobrado ahora ' + fm(montoAhora) + ', saldo ' + fm(saldoPendiente) + ')' : ''));
  });
}
var _ventasVisible = 20;
function renVentas(resetPag) {
  if (resetPag) _ventasVisible = 20;
  var desdeEl = el('vDesde'), hastaEl = el('vHasta');
  var desde = desdeEl && desdeEl.value ? new Date(desdeEl.value + 'T00:00:00') : null;
  var hasta = hastaEl && hastaEl.value ? new Date(hastaEl.value + 'T23:59:59') : null;
  var buscar = (el('bVentas') ? (el('bVentas').value || '') : '').toLowerCase().trim();
  var vs = state.ventas.filter(function(v) {
    if (desde && new Date(v.fecha + 'T12:00:00') < desde) return false;
    if (hasta && new Date(v.fecha + 'T12:00:00') > hasta) return false;
    if (buscar && String(v.nombre || '').toLowerCase().indexOf(buscar) < 0) return false;
    return true;
  });
  // Ordenar de más reciente a más antigua (por fecha + hora; la última queda arriba aunque refresques)
  vs.sort(function(a, b) { var ka = ventaKey(a), kb = ventaKey(b); return kb > ka ? 1 : kb < ka ? -1 : 0; });
  if (el('ventasRes')) {
    var factFiltro = vs.reduce(function(s, v) { return s + (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0); }, 0);
    el('ventasRes').textContent = vs.length + ' venta' + (vs.length !== 1 ? 's' : '') + (vs.length ? ' · ' + fm(factFiltro) : '');
  }
  if (!vs.length) { el('tVentas').innerHTML = '<div class="empty">Sin ventas</div>'; return; }
  var total = vs.length;
  var visibles = vs.slice(0, _ventasVisible);
  var filas = visibles.map(function(v) {
    var m = (parseFloat(v.costoUnit) || 0) > 0 ? ((parseFloat(v.precioVenta) - parseFloat(v.costoUnit)) / parseFloat(v.costoUnit) * 100) : 0;
    return '<tr><td><div>' + v.nombre + '</div><div style="font-size:11px;color:var(--muted)">' + fd(v.fecha) + (v.hora ? ' · ' + v.hora : '') + '</div></td>' +
      '<td class="mono">' + v.cantidad + '</td>' +
      '<td class="mono ts">' + fm((parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0)) + '</td>' +
      '<td class="mono ' + (m >= 80 ? 'ts' : 'tw') + '">' + m.toFixed(0) + '%</td>' +
      '<td>' + (v.medio2
        ? '<span class="tag tga" title="' + fm(v.montoMedio1) + ' ' + (v.medio1 || v.medio) + ' + ' + fm(v.montoMedio2) + ' ' + v.medio2 + '">' + (v.medio1 || v.medio) + ' + ' + v.medio2 + '</span>'
        : '<select class="esel" onchange="updMedioVenta(\'' + v.id + '\',this.value)">' +
          ['Efectivo','Transferencia','Débito','Crédito','Pago QR'].map(function(m) { return '<option' + (v.medio === m ? ' selected' : '') + '>' + m + '</option>'; }).join('') +
          '</select>') +
      '</td>' +
      '<td><button class="btn btn-sm" style="background:#1f0000;color:var(--danger);padding:4px 8px" onclick="borrarVenta(\'' + v.id + '\')">✕</button></td></tr>';
  }).join('');
  var verMas = '';
  if (total > _ventasVisible) {
    verMas = '<div style="padding:14px;text-align:center;border-top:1px solid var(--border)">' +
      '<button class="btn btn-g btn-sm" onclick="_ventasVisible+=20;renVentas()">Ver más (' + (total - _ventasVisible) + ' restantes)</button>' +
      '</div>';
  }
  el('tVentas').innerHTML = '<table><thead><tr><th>Producto</th><th>Uds.</th><th>Total</th><th>Margen</th><th>Medio</th><th></th></tr></thead><tbody>' + filas + '</tbody></table>' + verMas;
  renCobrosPendientes();
}

// COBROS PENDIENTES (ventas con cobro parcial cuyo saldo todavía no se cobró)

  if (!el('cobrosPendCard')) return;

  var pend = state.ventas.filter(function(v) { return (parseFloat(v.saldoPendiente) || 0) > 0 && !v.cobradoSaldo; });

  if (!pend.length) return;

  el('cobrosPend').innerHTML = pend.map(function(v) {
    var vencida = v.fechaRecordatorio && v.fechaRecordatorio <= hoy;
    var wa = waLinkSaldo(v.clienteWA, v.cliente, v.saldoPendiente);
    return '<div style="display:flex;justify-content:space-between;align-items:flex-start;gap:10px;padding:10px 0;border-bottom:1px solid var(--border);flex-wrap:wrap">' +

        '<div style="font-weight:600">' + (v.cliente || '(sin nombre)') + '</div>' +

      '</div>' +

function cobrarSaldo(id) {
  var v = state.ventas.filter(function(x) { return x.id === id; })[0];
  if (!v) return;
  var sel = el('ms_' + id);
  var medio = sel ? sel.value : (v.medio || 'Efectivo');
  if (!confirm('¿Confirmar el cobro del saldo de ' + fm(v.saldoPendiente) + ' (' + medio + ')?')) return;
  var hoy = today();
  apiPost({ accion: 'actualizar', hoja: 'ventas', id: id, campo: 'cobradoSaldo', valor: true }, function() {});
  apiPost({ accion: 'actualizar', hoja: 'ventas', id: id, campo: 'fechaCobroSaldo', valor: hoy }, function() {});
  apiPost({ accion: 'actualizar', hoja: 'ventas', id: id, campo: 'medioSaldo', valor: medio }, function(err) {
    if (err) return alert('Error al registrar el cobro.');
    v.cobradoSaldo = true; v.fechaCobroSaldo = hoy; v.medioSaldo = medio;
    renVentas(); renDash();
    showMsg('msgV', 's', '✓ Saldo cobrado — ' + fm(v.saldoPendiente));
  });
}
// Mensaje de WhatsApp recordando un saldo pendiente de una venta con cobro parcial
function waLinkSaldo(wa, nombre, saldo) {
  if (!wa) return null;
  var digits = String(wa).replace(/\D/g, '');
  if (digits.charAt(0) === '0') digits = digits.slice(1);
  if (digits.slice(0, 2) !== '54') digits = '54' + digits;
  var msg = encodeURIComponent('Hola ' + (nombre || '') + '! Te escribimos de Tecnopuerto para recordarte el saldo pendiente de ' + fm(saldo) + '. ¡Gracias!');
  return 'https://wa.me/' + digits + '?text=' + msg;
}

function renGastos() {
  if (el('gFecha') && !el('gFecha').value) el('gFecha').value = today();
  var buscar = (el('bGastos') ? (el('bGastos').value || '') : '').toLowerCase();
  var arr = state.gastos.slice()
    .filter(function(g) { return !buscar || String(g.detalle || '').toLowerCase().indexOf(buscar) >= 0 || String(g.categoria || '').toLowerCase().indexOf(buscar) >= 0; })
    .sort(function(a, b) { return String(b.fecha || '').localeCompare(String(a.fecha || '')); });
  var mes = today().slice(0, 7);
  var delMes = state.gastos.filter(function(g) { return String(g.fecha).slice(0, 7) === mes; });
  var totMes = delMes.reduce(function(s, g) { return s + (parseFloat(g.monto) || 0); }, 0);
  var porCat = {};
  delMes.forEach(function(g) { var c = g.categoria || 'Otros'; porCat[c] = (porCat[c] || 0) + (parseFloat(g.monto) || 0); });
  if (el('gResumen')) {
    var detCat = Object.entries(porCat).sort(function(a, b) { return b[1] - a[1]; }).map(function(x) { return x[0] + ' ' + fm(x[1]); }).join(' · ');
    el('gResumen').innerHTML = 'Este mes: <span class="mono tw">' + fm(totMes) + '</span>' + (detCat ? ' · ' + detCat : '');
  }
  if (!el('tGastos')) return;
  if (!arr.length) { el('tGastos').innerHTML = '<div class="empty">Sin gastos cargados</div>'; return; }
  el('tGastos').innerHTML = '<table><thead><tr><th>Fecha</th><th>Categoría</th><th>Detalle</th><th>Monto</th><th>Medio</th><th></th></tr></thead><tbody>' +
    arr.map(function(g) {
      return '<tr>' +
        '<td class="mono" style="font-size:11px;color:var(--muted)">' + fd(g.fecha) + '</td>' +
        '<td><select class="esel" onchange="updGasto(\'' + g.id + '\',\'categoria\',this.value)">' + GASTO_CATS.map(function(c) { return '<option' + (g.categoria === c ? ' selected' : '') + '>' + c + '</option>'; }).join('') + '</select></td>' +
        '<td><input class="esel" style="width:170px;font-size:12px" value="' + String(g.detalle || '').replace(/"/g, '&quot;') + '" onblur="updGasto(\'' + g.id + '\',\'detalle\',this.value)" onkeydown="if(event.key===\'Enter\')this.blur()"></td>' +
        '<td class="mono ts"><input type="number" class="esel" style="width:90px" value="' + (parseFloat(g.monto) || 0) + '" onblur="updGasto(\'' + g.id + '\',\'monto\',this.value)" onkeydown="if(event.key===\'Enter\')this.blur()"></td>' +
        '<td><select class="esel" onchange="updGasto(\'' + g.id + '\',\'medio\',this.value)">' + ['Efectivo', 'Transferencia', 'Débito', 'Crédito', 'Pago QR'].map(function(m) { return '<option' + (g.medio === m ? ' selected' : '') + '>' + m + '</option>'; }).join('') + '</select></td>' +
        '<td><button class="btn btn-sm" style="background:#1f0000;color:var(--danger);padding:4px 8px" onclick="borrarGasto(\'' + g.id + '\')">✕</button></td></tr>';
    }).join('') + '</tbody></table>';
}
function updGasto(id, campo, valor) {
  var g = state.gastos.filter(function(x) { return x.id === id; })[0];
  if (!g) return;
  if (campo === 'monto') valor = parseFloat(valor) || 0;
  if (g[campo] === valor) return;
  g[campo] = valor;
  apiPost({ accion: 'actualizar', hoja: 'gastos', id: id, campo: campo, valor: valor }, function(err) {
    if (!err) { renGastos(); renStats(); }
  });
}
function borrarGasto(id) {
  if (!confirm('¿Borrar este gasto?')) return;
  apiPost({ accion: 'borrar', hoja: 'gastos', id: id }, function(err) {
    if (err) return alert('Error al borrar.');
    state.gastos = state.gastos.filter(function(g) { return g.id !== id; });
    renGastos(); renStats();
  });
}
function borrarVenta(id) {
  var venta = state.ventas.filter(function(v) { return v.id === id; })[0];
  if (!venta) return;
  // Motivo OPCIONAL: si cancela el prompt no se borra; si deja vacío, se borra igual
  var motivo = prompt('Borrar venta de "' + venta.nombre + '".\nMotivo (opcional, podés dejarlo vacío):', '');
  if (motivo === null) return;
  apiPost({ accion: 'borrarVenta', id: id, motivo: motivo }, function(err) {
    if (err) return alert('Error al borrar.');
    if (venta) {
      var prod = state.stock.filter(function(s) {
        return String(s.nombre).toUpperCase().trim() === String(venta.nombre).toUpperCase().trim();
      })[0];
      if (prod) {
        var nuevaCant = (parseInt(prod.cantidad) || 0) + (parseInt(venta.cantidad) || 0);
        apiPost({ accion: 'actualizarStockPorNombre', nombre: venta.nombre, valor: nuevaCant }, function() {});
        prod.cantidad = nuevaCant;
      }
    }
    state.ventas = state.ventas.filter(function(v) { return v.id !== id; });
    renVentas(); renDash(); renStock(); renSelProd();
  });
}

function editarRep(id) {
  var r = state.reparaciones.filter(function(x) { return x.id === id; })[0];
  if (!r) return;
  editandoRepId = id;
  el('rCli').value = r.cliente || '';
  el('rWA').value = r.whatsapp || '';
  el('rEq').value = r.equipo || '';
  el('rProb').value = r.problema || '';
  el('rPrc').value = parseFloat(r.precioReparacion) || 0;
  el('rCos').value = parseFloat(r.costoRepuesto) || 0;
  el('rNot').value = r.notas || '';
  if (el('btnGuardarRep')) el('btnGuardarRep').textContent = 'Guardar cambios';
  el('fRep').style.display = 'block';
  el('fRep').scrollIntoView({ behavior: 'smooth', block: 'start' });
}
function guardarRep() {
  // Precio OPCIONAL a propósito: no siempre se puede presupuestar al toque (modelo fuera de la
  // lista de precios, hay que consultar al técnico). Sin precio, la reparación entra en
  // "Esperando presupuesto" en vez de "En proceso" — ver renRepEspera / [[crm-reparaciones-ia-agentica]].
  var campos = { cliente: el('rCli').value, whatsapp: el('rWA').value, equipo: el('rEq').value, problema: el('rProb').value, precioReparacion: parseFloat(el('rPrc').value) || 0, costoRepuesto: parseFloat(el('rCos').value) || 0, tecnico: el('rTec').value, notas: el('rNot').value };
  if (!campos.cliente || !campos.equipo) return alert('Completá cliente y equipo.');

  if (editandoRepId) {
    var id = editandoRepId;
    var r = state.reparaciones.filter(function(x) { return x.id === id; })[0];
    if (!r) return;
    // Si estaba esperando presupuesto y ahora se cargó un precio, pasa sola a "En proceso".
    if (r.estado === 'Esperando presupuesto' && campos.precioReparacion > 0) campos.estado = 'En proceso';
    Object.keys(campos).forEach(function(campo) {
      if (r[campo] === campos[campo]) return;
      r[campo] = campos[campo];
      apiPost({ accion: 'actualizar', hoja: 'reparaciones', id: id, campo: campo, valor: campos[campo] }, function(err) { if (err) console.error('Error al actualizar ' + campo); });
    });
    editandoRepId = null;
    if (el('btnGuardarRep')) el('btnGuardarRep').textContent = 'Guardar reparación';
    renRep(); renDash();
    ['rCli', 'rWA', 'rEq', 'rProb', 'rPrc', 'rNot'].forEach(function(id2) { el(id2).value = ''; });
    el('rCos').value = '0'; el('fRep').style.display = 'none';
    return;
  }

  var rep = { id: gid(), fecha: today(), estado: campos.precioReparacion > 0 ? 'En proceso' : 'Esperando presupuesto' };
  Object.keys(campos).forEach(function(k) { rep[k] = campos[k]; });
  apiPost({ accion: 'agregar', hoja: 'reparaciones', item: rep }, function(err) {
    if (err) return alert('Error al guardar.');
    state.reparaciones.unshift(rep);
    renRep(); renDash();
    ['rCli', 'rWA', 'rEq', 'rProb', 'rPrc', 'rNot'].forEach(function(id) { el(id).value = ''; });
    el('rCos').value = '0'; el('fRep').style.display = 'none';
  });
}

function updEstado(id, estado) {
  var rPrev = state.reparaciones.filter(function(x) { return x.id === id; })[0];
  // Entregado pide medio de pago antes de guardar nada — ver abrirModalEntrega/confirmarEntrega.
  // Si ya estaba Entregado (re-selección sin cambio real), no reabre el modal.
  if (estado === 'Entregado') {
    if (rPrev && rPrev.estado !== 'Entregado') {
      if (!(parseFloat(rPrev.costoRepuesto) > 0)) {
        alert('Cargá el costo técnico antes de marcar como Entregado.');
        renRep();
        return;
      }
      abrirModalEntrega(id);
    }
    return;
  }
  // Se guarda cuándo pasó a "Listo" (solo la primera vez) para poder mostrar la antigüedad
  // en esa columna — cuánto hace que está esperando que la retiren, no cuánto hace que ingresó.
  var pasaAListo = estado === 'Listo' && rPrev && rPrev.estado !== 'Listo';
  apiPost({ accion: 'actualizar', hoja: 'reparaciones', id: id, campo: 'estado', valor: estado }, function(err) {
    if (err) return alert('Error.');
    var r = state.reparaciones.filter(function(x) { return x.id === id; })[0];
    if (r) r.estado = estado;
    if (pasaAListo) {
      var hoyStr2 = today();
      if (r) r.fechaListo = hoyStr2;
      apiPost({ accion: 'actualizar', hoja: 'reparaciones', id: id, campo: 'fechaListo', valor: hoyStr2 }, function() {});
    }
    renRep(); renDash();
  });
}

// Al cancelar no se tocó el estado en `state`, así que basta con re-renderizar para que
// el <select> de estado vuelva a mostrar el valor real (no el que el usuario acaba de tocar).
function cancelarEntrega() {
  modalEntregaId = null;
  el('modalEntrega').style.display = 'none';
  renRep();
}
function toggleEmCombinado() {
  el('emComboBox').style.display = el('emCombinado').checked ? 'grid' : 'none';
  calcEmTotal();
}
function calcEmTotal() {
  var r = state.reparaciones.filter(function(x) { return x.id === modalEntregaId; })[0];
  if (!r || !el('emCombinado').checked) { if (el('emResumen')) el('emResumen').textContent = ''; return; }
  var total = parseFloat(r.precioReparacion) || 0;
  var m2 = parseFloat(el('emMonto2').value) || 0;
  var m1 = total - m2;
  el('emResumen').textContent = m1 >= 0 ? (fm(m1) + ' + ' + fm(m2) + ' = ' + fm(total)) : 'El segundo monto no puede superar el total (' + fm(total) + ')';
  el('emResumen').style.color = m1 >= 0 ? '' : 'var(--danger)';
}
function confirmarEntrega() {
  var id = modalEntregaId;
  var r = state.reparaciones.filter(function(x) { return x.id === id; })[0];
  if (!r) return cancelarEntrega();
  var medio1 = el('emMedio1').value;
  if (!medio1) return alert('Elegí el medio de pago.');
  var combinado = el('emCombinado').checked;
  var total = parseFloat(r.precioReparacion) || 0;
  var medio2 = '', montoMedio1 = '', montoMedio2 = '';
  if (combinado) {
    medio2 = el('emMedio2').value;
    if (!medio2) return alert('Elegí el segundo medio de pago.');
    montoMedio2 = parseFloat(el('emMonto2').value) || 0;
    montoMedio1 = total - montoMedio2;
    if (montoMedio2 <= 0 || montoMedio1 < 0) return alert('Revisá los montos: no pueden superar el total (' + fm(total) + ').');
  }
  // Reseña por WhatsApp — misma lógica de antes (primera vez que pasa a Entregado), disparada
  // sincrónica con el click para que el navegador no la bloquee como popup.
  if (r.estado !== 'Entregado' && r.whatsapp) window.open(waLinkResena(r.whatsapp, r.cliente), '_blank');
  el('modalEntrega').style.display = 'none';
  var hoyStr = today();
  var campos = { estado: 'Entregado', fechaEntrega: hoyStr, medio: medio1, medio1: medio1, montoMedio1: montoMedio1, medio2: medio2, montoMedio2: montoMedio2 };
  var pendientes = Object.keys(campos).length;
  Object.keys(campos).forEach(function(campo) {
    apiPost({ accion: 'actualizar', hoja: 'reparaciones', id: id, campo: campo, valor: campos[campo] }, function(err) {
      if (!err) r[campo] = campos[campo];
      pendientes--;
      if (pendientes === 0) { modalEntregaId = null; renRep(); renDash(); renStats(); }
    });
  });
}
// Filtro compartido de Reparaciones (usado por renRep y exportRepXL, para que la
// exportación siempre respete exactamente lo que se ve en pantalla)
function _filtrarReparaciones() {
  var buscar = (el('bRep') ? (el('bRep').value || '') : '').toLowerCase();
  var desde = el('rDesde') && el('rDesde').value ? el('rDesde').value : null;
  var hasta = el('rHasta') && el('rHasta').value ? el('rHasta').value : null;
  var ord = { 'Esperando presupuesto': 0, 'En proceso': 1, 'Listo': 2, 'Abandonado': 3, 'Entregado': 4 };
  return state.reparaciones
    .filter(function(r) { return filtroRep === 'todos' || (filtroRep === 'activas' ? (r.estado === 'Esperando presupuesto' || r.estado === 'En proceso' || r.estado === 'Listo') : r.estado === filtroRep); })
    .filter(function(r) { if (desde && r.fecha < desde) return false; if (hasta && r.fecha > hasta) return false; return true; })
    .filter(function(r) {
      if (!buscar) return true;
      if (r.cliente.toLowerCase().indexOf(buscar) >= 0 || r.equipo.toLowerCase().indexOf(buscar) >= 0) return true;
      if ((r.tecnico || '').toLowerCase().indexOf(buscar) >= 0) return true;
      var precio = String(parseFloat(r.precioReparacion) || 0);
      var costo = String(parseFloat(r.costoRepuesto) || 0);
      return precio.indexOf(buscar) >= 0 || costo.indexOf(buscar) >= 0;
    })
    .slice().sort(function(a, b) { return (ord[a.estado] !== undefined ? ord[a.estado] : 99) - (ord[b.estado] !== undefined ? ord[b.estado] : 99); });
}
function exportRepXL() {
  var fil = _filtrarReparaciones();
  if (!fil.length) { alert('No hay reparaciones para exportar con los filtros actuales.'); return; }
  var wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(
    [['Fecha', 'Cliente', 'WhatsApp', 'Equipo', 'Problema', 'Precio', 'Costo técnico', 'Ganancia', 'Recibió', 'Estado', 'Medio', 'Notas']].concat(
      fil.map(function(r) { return [r.fecha, r.cliente, r.whatsapp || '', r.equipo, r.problema, parseFloat(r.precioReparacion) || 0, parseFloat(r.costoRepuesto) || 0, (parseFloat(r.precioReparacion) || 0) - (parseFloat(r.costoRepuesto) || 0), r.tecnico, r.estado, _medioLabelRep(r), r.notas || '']; })
    )), 'Reparaciones');
  XLSX.writeFile(wb, 'Tecnopuerto_Reparaciones_' + today() + '.xlsx');
}

// Vista "Activas": En proceso y Listo en columnas separadas (en vez de una sola lista
// larga con scroll) — ver [[crm-reparaciones-activas]]. `fil` ya viene filtrado a
// {En proceso, Listo} por _filtrarReparaciones().
function renRepActivas(fil, eC) {
  var espera = fil.filter(function(r) { return r.estado === 'Esperando presupuesto'; });
  var proc = fil.filter(function(r) { return r.estado === 'En proceso'; });
  var listo = fil.filter(function(r) { return r.estado === 'Listo'; });
  renRepEspera(espera, eC);
  renRepMini('tRepProc', proc, eC, false);
  renRepMini('tRepListo', listo, eC, true);
  if (el('repEsperaRes')) el('repEsperaRes').textContent = espera.length + (espera.length === 1 ? ' reparación' : ' reparaciones');
  if (el('repProcRes')) el('repProcRes').textContent = proc.length + (proc.length === 1 ? ' reparación' : ' reparaciones');

// Antigüedad de una reparación en su etapa actual: días desde que ingresó (columna "En proceso")
// o desde que quedó lista (columna "Listo", usa fechaListo — con fallback a fecha para registros
// viejos de antes de este cambio, que no la tienen). Se resalta en amarillo/rojo si se estanca,
// para que no se pierdan de vista en el embrollo cuando hay varias en simultáneo.
function _diasDesde(fechaStr) {
  if (!fechaStr) return null;
  var d = new Date(fechaStr + 'T12:00:00');
  var hoy = new Date(today() + 'T12:00:00');
  return Math.round((hoy - d) / 86400000);
}

function renReposicion() {
  var cont = el('reposicion');
  if (!cont) return;
  var rep = state.stock.filter(function(s) {
    var c = parseInt(s.cantidad) || 0;
    var m = parseInt(s.stockMinimo) || 3;
    return c <= m;
  }).sort(function(a, b) { return (parseInt(a.cantidad) || 0) - (parseInt(b.cantidad) || 0); });
  var btn = el('btnRepo');
  if (!rep.length) { cont.style.display = 'none'; cont.innerHTML = ''; if (btn) btn.style.display = 'none'; return; }
  if (btn) { btn.style.display = ''; btn.textContent = '🛒 Reposición (' + rep.length + ')'; }
  // Velocidad de venta real de los últimos 30 días: sugiere en base a lo que efectivamente
  // se vende (cubrir DIAS_COBERTURA de demanda esperada), no una fórmula fija arbitraria.
  var DIAS_COBERTURA = 14;
  var hace30 = new Date(); hace30.setDate(hace30.getDate() - 30);
  var fechaCorte = localDateStr(hace30);
  var ventas30 = {};
  state.ventas.forEach(function(v) {
    if (v.fecha >= fechaCorte) ventas30[v.nombre] = (ventas30[v.nombre] || 0) + (parseInt(v.cantidad) || 0);
  });
  cont.innerHTML = '<h3 style="margin-bottom:4px">🛒 Reposición sugerida — ' + rep.length + ' producto' + (rep.length !== 1 ? 's' : '') + ' para comprar</h3>' +
    '<div class="tm" style="font-size:11px;margin-bottom:12px">Basada en lo vendido los últimos 30 días, para cubrir ' + DIAS_COBERTURA + ' días de demanda. Sin historial de venta, usa el mínimo como referencia.</div>' +
    '<table><thead><tr><th>Producto</th><th>Stock</th><th>Mín.</th><th>Vendido/30d</th><th>Comprar (sug.)</th><th>Estado</th></tr></thead><tbody>' +
    rep.map(function(s) {
      var c = parseInt(s.cantidad) || 0;
      var m = parseInt(s.stockMinimo) || 3;
      var vendido = ventas30[s.nombre] || 0;
      var objetivo = Math.ceil((vendido / 30) * DIAS_COBERTURA);
      var sug = (objetivo - c) > 0 ? (objetivo - c) : Math.max(m - c, 1);
      var cls = c === 0 ? 'tgd' : 'tgw';
      var lbl = c === 0 ? 'Sin stock' : 'Bajo';
      return '<tr><td>' + s.nombre + '</td><td class="mono">' + c + '</td><td class="mono tm">' + m + '</td>' +
        '<td class="mono tm">' + (vendido || '—') + '</td>' +
        '<td class="mono ts">' + sug + '</td><td><span class="tag ' + cls + '">' + lbl + '</span></td></tr>';
    }).join('') + '</tbody></table>';
}

function impExcel(e) {
  var file = e.target.files[0]; if (!file) return;
  var reader = new FileReader();
  reader.onload = function(evt) {
    try {
      var wb = XLSX.read(evt.target.result, { type: 'binary' });
      var ws = wb.Sheets[wb.SheetNames[0]];
      var raw = XLSX.utils.sheet_to_json(ws, { header: 1 });

      // Buscar fila de encabezados
      var hIdx = -1;
      for (var i = 0; i < raw.length; i++) {
        if (Array.isArray(raw[i]) && raw[i].some(function(c) { return String(c || '').toLowerCase() === 'articulo'; })) { hIdx = i; break; }
      }
      if (hIdx === -1) { showMsg('msgS', 'd', 'No se encontró columna "Articulo" en el Excel.'); return; }

      var heads = (raw[hIdx] || []).map(function(h) { return String(h || '').trim().toLowerCase(); });
      var iN = heads.indexOf('articulo');
      var iS = heads.indexOf('stock');
      var iC = heads.indexOf('neto');   // columna de costo
      var iA = heads.indexOf('activo');
      var iSv = heads.indexOf('servicio');

      if (iN === -1 || iS === -1 || iC === -1) {
        showMsg('msgS', 'd', 'No se encontraron columnas Articulo / Stock / Neto.');
        return;
      }

      impPrev = raw.slice(hIdx + 1)
        .filter(function(r) {
          var activo = iA === -1 || String(r[iA] || '').toUpperCase() === 'SI';
          var esServicio = iSv !== -1 && String(r[iSv] || '').toUpperCase() === 'SI';
          var tieneStock = (parseFloat(r[iS]) || 0) > 0;
          return activo && !esServicio && tieneStock && r[iN];
        })
        .map(function(r) {
          return { id: gid(), nombre: String(r[iN]).toUpperCase().trim(), cantidad: Math.round(parseFloat(r[iS]) || 0), costo: Math.round(parseFloat(r[iC]) || 0) };
        });

      var nuevos = impPrev.filter(function(i) {
        return !state.stock.filter(function(s) { return s.nombre === i.nombre; }).length;
      }).length;

      el('impInfo').textContent = impPrev.length + ' productos · ' + nuevos + ' nuevos · ' + (impPrev.length - nuevos) + ' a actualizar (suma stock)';
      el('impBody').innerHTML = impPrev.slice(0, 60).map(function(i) {
        return '<tr><td style="font-size:12px">' + i.nombre + '</td><td class="mono">' + i.cantidad + '</td><td class="mono">' + fm(i.costo) + '</td></tr>';
      }).join('');
      if (impPrev.length > 60) el('impBody').innerHTML += '<tr><td colspan="3" class="tm" style="font-size:12px;text-align:center">...y ' + (impPrev.length - 60) + ' más</td></tr>';
      el('modImp').style.display = 'flex';
    } catch (err) { showMsg('msgS', 'd', 'Error leyendo el Excel.'); }
  };
  reader.readAsBinaryString(file); e.target.value = '';
}
function impPrecios(e) {
  var file = e.target.files[0]; if (!file) return;
  var reader = new FileReader();
  reader.onload = function(evt) {
    try {
      var wb = XLSX.read(evt.target.result, { type: 'binary' });
      var ws = wb.Sheets[wb.SheetNames[0]];
      var raw = XLSX.utils.sheet_to_json(ws, { header: 1 });

      var hIdx = -1;
      for (var i = 0; i < raw.length; i++) {
        if (Array.isArray(raw[i]) && raw[i].some(function(c) { return String(c || '').toLowerCase().trim() === 'nombre'; })) { hIdx = i; break; }
      }
      if (hIdx === -1) { showMsg('msgS', 'd', 'No se encontró columna "nombre" en el Excel.'); return; }

      var heads = (raw[hIdx] || []).map(function(h) { return String(h || '').trim().toLowerCase(); });
      var iN = heads.indexOf('nombre');
      var iPV = heads.indexOf('precioventa');
      if (iN === -1 || iPV === -1) { showMsg('msgS', 'd', 'Se necesitan columnas "nombre" y "precioVenta".'); return; }

      var items = raw.slice(hIdx + 1)
        .filter(function(r) { return r[iN] && (parseFloat(r[iPV]) || 0) > 0; })
        .map(function(r) { return { nombre: String(r[iN]).toUpperCase().trim(), precioVenta: Math.round(parseFloat(r[iPV]) || 0) }; });

      if (!items.length) { showMsg('msgS', 'd', 'No hay filas válidas para importar.'); return; }
      if (!confirm('¿Actualizar el precio de venta público de ' + items.length + ' productos existentes en el Stock?')) return;

      apiPost({ accion: 'importarPrecios', items: items }, function(err, d) {
        if (err) return showMsg('msgS', 'd', 'Error al importar precios.');
        items.forEach(function(item) {
          var s = state.stock.filter(function(x) { return x.nombre === item.nombre; })[0];
          if (s) s.precioVenta = item.precioVenta;

        renStock(); renSelProd();
        showMsg('msgS', 's', '✓ Precio de venta actualizado en ' + (d && d.actualizados != null ? d.actualizados : items.length) + ' productos.');
      });

  };
  reader.readAsBinaryString(file); e.target.value = '';

function renStats() {
  var desde = new Date(el('dDesde').value + 'T00:00:00');
  var hasta = new Date(el('dHasta').value + 'T23:59:59');
  function inR(f) { var d = new Date(f + 'T12:00:00'); return d >= desde && d <= hasta; }
  // Período anterior de igual duración, inmediatamente antes del período elegido — para
  // mostrar variación ↑/↓ en las tarjetas KPI de Reparaciones, Ganancia neta y Ticket promedio.
  var prevHasta = new Date(desde.getTime() - 1000);
  var prevDesde = new Date(prevHasta.getTime() - (hasta.getTime() - desde.getTime()));
  function inPrev(f) { var d = new Date(f + 'T12:00:00'); return d >= prevDesde && d <= prevHasta; }
  var rFp = state.reparaciones.filter(function(r) { return inPrev(r.fecha); });
  var vFp = state.ventas.filter(function(v) { return inPrev(v.fecha); });
  var rEntp = state.reparaciones.filter(function(r) { return r.estado === 'Entregado' && inPrev(r.fechaEntrega || r.fecha); });
  var tVp = vFp.reduce(function(s, v) { return s + (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0); }, 0);
  var cVp = vFp.reduce(function(s, v) { return s + (parseFloat(v.costoUnit) || 0) * (parseInt(v.cantidad) || 0); }, 0);
  var tRp = rEntp.reduce(function(s, r) { return s + (parseFloat(r.precioReparacion) || 0); }, 0);
  var cRp = rEntp.reduce(function(s, r) { return s + (parseFloat(r.costoRepuesto) || 0); }, 0);
  var opsP = vFp.length + rEntp.length;
  var ticketP = opsP > 0 ? (tVp + tRp) / opsP : 0;
  var gastosFp = (state.gastos || []).filter(function(g) { return inPrev(g.fecha) && g.categoria !== 'Mercadería/Stock'; });
  var totGastosP = gastosFp.reduce(function(s, g) { return s + (parseFloat(g.monto) || 0); }, 0);
  var gananciaNetaP = ((tVp - cVp) + (tRp - cRp)) - totGastosP;
  var vF = state.ventas.filter(function(v) { return inR(v.fecha); });
  var rF = state.reparaciones.filter(function(r) { return inR(r.fecha); });
  var tV = vF.reduce(function(s, v) { return s + (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0); }, 0);
  var cV = vF.reduce(function(s, v) { return s + (parseFloat(v.costoUnit) || 0) * (parseInt(v.cantidad) || 0); }, 0);
  var gV = tV - cV, mV = cV > 0 ? (gV / cV * 100) : 0;
  var rEnt = state.reparaciones.filter(function(r) { return r.estado === 'Entregado' && inR(r.fechaEntrega || r.fecha); });
  var tR = rEnt.reduce(function(s, r) { return s + (parseFloat(r.precioReparacion) || 0); }, 0);
  var cR = rEnt.reduce(function(s, r) { return s + (parseFloat(r.costoRepuesto) || 0); }, 0);
  var gR = tR - cR;
  el('stT').textContent = fm(tV + tR); el('stG').textContent = fm(gV + gR);
  el('stMs').textContent = 'Incl. reparaciones · margen accesorios: ' + mV.toFixed(0) + '%';
  el('stR').textContent = rF.length; el('stRs').textContent = fm(tR) + ' · ' + rEnt.length + ' entregadas';
  renderDelta('stRDelta', rF.length, rFp.length);
  el('stGR').textContent = fm(gR); el('stCR').textContent = 'Pagado a técnicos: ' + fm(cR);
  el('stV').textContent = fm(tV); el('stRv').textContent = fm(tR); el('stTot').textContent = fm(tV + tR);
  var medios = {};
  vF.forEach(function(v) { medioPartes(v).forEach(function(p) { medios[p.medio] = (medios[p.medio] || 0) + p.monto; }); });
  rEnt.forEach(function(r) { repMedioPartes(r).forEach(function(p) { medios[p.medio] = (medios[p.medio] || 0) + p.monto; }); });
  var totMedios = tV + tR;
  el('tMedios').innerHTML = Object.keys(medios).length === 0 ? '<div class="empty">Sin datos</div>' :
    '<table><thead><tr><th>Medio</th><th>Total</th><th>%</th></tr></thead><tbody>' +
    Object.entries(medios).sort(function(a, b) { return b[1] - a[1]; })
      .map(function(x) { return '<tr><td>' + x[0] + '</td><td class="mono ts">' + fm(x[1]) + '</td><td class="mono">' + (totMedios > 0 ? (x[1] / totMedios * 100).toFixed(0) : 0) + '%</td></tr>'; }).join('') +
    '</tbody></table>';
  var dias = {};
  for (var i = 13; i >= 0; i--) { var d = new Date(); d.setDate(d.getDate() - i); var k = localDateStr(d); dias[k] = { v: 0, r: 0, label: d.toLocaleDateString('es-AR', { weekday: 'short', day: 'numeric' }), labelFull: d.toLocaleDateString('es-AR', { weekday: 'long', day: 'numeric', month: 'long' }) }; }
  state.ventas.forEach(function(v) { if (dias[v.fecha]) dias[v.fecha].v += (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0); });
  state.reparaciones.forEach(function(r) { var fd = r.fechaEntrega || r.fecha; if (r.estado === 'Entregado' && dias[fd]) dias[fd].r += (parseFloat(r.precioReparacion) || 0); });
  // Encontrar el día pico para resaltarlo (motiva ver el récord de un vistazo)
  var diasArr = Object.entries(dias);
  var diasConVentas = diasArr.filter(function(x) { return (x[1].v + x[1].r) > 0; });
  var promedio = diasConVentas.length ? diasConVentas.reduce(function(s, x) { return s + x[1].v + x[1].r; }, 0) / diasConVentas.length : 0;
  var mejorKey = null, mejorTotal = 0;
  diasArr.forEach(function(x) { var t = x[1].v + x[1].r; if (t > mejorTotal) { mejorTotal = t; mejorKey = x[0]; } });
  if (el('mejorDiaBadge')) {
    if (mejorKey && mejorTotal > 0) {
      var vsProm = promedio > 0 ? Math.round((mejorTotal / promedio - 1) * 100) : 0;
      el('mejorDiaBadge').innerHTML = '🏆 Mejor día: <span style="color:var(--text);font-weight:600">' + dias[mejorKey].labelFull + '</span> — <span class="mono ts">' + fm(mejorTotal) + '</span>' + (vsProm > 0 ? ' <span class="tm">(+' + vsProm + '% vs. el promedio)</span>' : '');
    } else {
      el('mejorDiaBadge').innerHTML = '';
    }
  }
  var maxV = Math.max.apply(null, diasArr.map(function(x) { return x[1].v + x[1].r; }).concat([1]));
  el('barC').innerHTML = diasArr.map(function(x) {
    var k = x[0], d = x[1], total = d.v + d.r;
    var esMejor = k === mejorKey && total > 0;
    var hV = Math.max(d.v / maxV * 100, 0), hR = Math.max(d.r / maxV * 100, 0);

      (esMejor ? '<div style="font-size:11px;line-height:1">🏆</div>' : '') +
      (total > 0 ? '<div class="bval" style="' + (esMejor ? 'color:var(--warning);font-weight:700' : '') + '">' + fm(total).replace('$', '') + '</div>' : '') +
      '<div style="width:100%;display:flex;flex-direction:column;justify-content:flex-end;height:100%;gap:2px' + (esMejor ? ';filter:drop-shadow(0 0 4px var(--warning))' : '') + '">' +
      (d.r > 0 ? '<div style="width:100%;height:' + hR + '%;background:var(--success);border-radius:3px 3px 0 0;min-height:3px"></div>' : '') +
      (d.v > 0 ? '<div style="width:100%;height:' + hV + '%;background:' + (esMejor ? 'var(--warning)' : 'var(--accent)') + ';border-radius:' + (d.r > 0 ? '0' : '3px 3px 0 0') + ';min-height:3px"></div>' : '') +
      (total === 0 ? '<div style="width:100%;height:3px;background:var(--border);border-radius:3px"></div>' : '') +
      '</div><div class="blabel" style="' + (esMejor ? 'color:var(--warning);font-weight:700' : '') + '">' + d.label + '</div></div>';
  }).join('');

  // Ticket promedio y operaciones del período
  var ops = vF.length + rEnt.length;
  var facturado = tV + tR;
  if (el('stTicket')) el('stTicket').textContent = ops > 0 ? fm(facturado / ops) : '—';
  if (el('stTicketSub')) el('stTicketSub').textContent = ops > 0 ? fm(facturado) + ' en ' + ops + ' oper.' : 'Sin operaciones';
  if (el('stOps')) el('stOps').textContent = ops;
  renderDelta('stTicketDelta', ops > 0 ? facturado / ops : 0, ticketP);

  // Ganancia NETA del período = ganancia bruta − gastos operativos (SIN contar mercadería,
  // porque el costo de la mercadería ya está descontado en cada venta vía costoUnit — contarla
  // de nuevo acá sería descontarla dos veces). Se muestra aparte solo a modo informativo.
  var gastosMerc = (state.gastos || []).filter(function(g) { return inR(g.fecha) && g.categoria === 'Mercadería/Stock'; });
  var totMerc = gastosMerc.reduce(function(s, g) { return s + (parseFloat(g.monto) || 0); }, 0);
  var gastosF = (state.gastos || []).filter(function(g) { return inR(g.fecha) && g.categoria !== 'Mercadería/Stock'; });
  var totGastos = gastosF.reduce(function(s, g) { return s + (parseFloat(g.monto) || 0); }, 0);
  var gananciaNeta = (gV + gR) - totGastos;
  if (el('stGastos')) el('stGastos').textContent = fm(totGastos);
  if (el('stGastosSub')) el('stGastosSub').textContent = totMerc > 0 ? 'Compras de mercadería (aparte, ya en costoUnit): ' + fm(totMerc) : 'No incluye compras de mercadería';

  }
  renderDelta('stNetaDelta', gananciaNeta, gananciaNetaP);

  // Evolución mensual — últimos 12 meses (independiente del filtro de fechas)
  renStatsMensual();

  // Stock sin rotación: con stock y sin ventas en los últimos ROT_DIAS días.
  // Para cada producto se calcula además la fecha de la última venta real (no solo
  // si vendió o no dentro de la ventana), para poder mostrar días exactos + semáforo.
  var ROT_DIAS = 60;
  var corte = new Date(); corte.setDate(corte.getDate() - ROT_DIAS);
  var hoyDS = new Date();
  var ultimaVenta = {};
  state.ventas.forEach(function(v) {
    var k = String(v.nombre).toUpperCase().trim();
    var f = new Date(v.fecha + 'T12:00:00');
    if (!ultimaVenta[k] || f > ultimaVenta[k]) ultimaVenta[k] = f;
  });
  var dead = state.stock.filter(function(s) {
    var k = String(s.nombre).toUpperCase().trim();
    return (parseInt(s.cantidad) || 0) > 0 && (!ultimaVenta[k] || ultimaVenta[k] < corte);
  }).map(function(s) {
    var k = String(s.nombre).toUpperCase().trim();
    var dias = ultimaVenta[k] ? Math.round((hoyDS - ultimaVenta[k]) / 86400000) : null;
    return { s: s, dias: dias };
  }).sort(function(a, b) { return ((parseFloat(b.s.costo) || 0) * (parseInt(b.s.cantidad) || 0)) - ((parseFloat(a.s.costo) || 0) * (parseInt(a.s.cantidad) || 0)); });
  if (el('tDeadStock')) {
    el('tDeadStock').innerHTML = !dead.length ? '<div class="empty">Todo el stock con stock tuvo ventas recientes 👍</div>' :
      '<table><thead><tr><th>Producto</th><th>Stock</th><th>Días sin venta</th><th>Estado</th><th>Costo unit.</th><th>Capital inmovilizado</th></tr></thead><tbody>' +
      dead.map(function(x) {

        var estado = (dias === null || dias >= 90) ? { tag: 'tgd', label: 'Crítico' } :
                     (dias >= ROT_DIAS) ? { tag: 'tgw', label: 'Atención' } : { tag: 'tgs', label: 'OK' };
        return '<tr><td>' + s.nombre + '</td><td class="mono">' + c + '</td><td class="mono">' + diasLabel + '</td>' +

  }

// Sparklines de tendencia (últimos 12 meses, independiente del filtro de fechas) para
// las tarjetas KPI de Ticket promedio, Ganancia neta y Reparaciones — ver [[crm-estadisticas-rediseno]]
function renKpiSparklines() {
  var meses = {};
  var hoy = new Date();
  for (var i = 11; i >= 0; i--) {
    var dM = new Date(hoy.getFullYear(), hoy.getMonth() - i, 1);
    var k = dM.getFullYear() + '-' + ('0' + (dM.getMonth() + 1)).slice(-2);
    meses[k] = { fact: 0, bruta: 0, gastos: 0, ops: 0, repEnt: 0 };
  }
  state.ventas.forEach(function(v) {
    var k = String(v.fecha || '').slice(0, 7);
    if (!meses[k]) return;
    var t = (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0);
    var c = (parseFloat(v.costoUnit) || 0) * (parseInt(v.cantidad) || 0);
    meses[k].fact += t; meses[k].bruta += (t - c); meses[k].ops += 1;
  });
  state.reparaciones.forEach(function(r) {
    if (r.estado !== 'Entregado') return;
    var k = String(r.fechaEntrega || r.fecha || '').slice(0, 7);
    if (!meses[k]) return;
    var t = parseFloat(r.precioReparacion) || 0, c = parseFloat(r.costoRepuesto) || 0;
    meses[k].fact += t; meses[k].bruta += (t - c); meses[k].ops += 1; meses[k].repEnt += 1;
  });
  (state.gastos || []).forEach(function(g) {
    if (g.categoria === 'Mercadería/Stock') return;
    var k = String(g.fecha || '').slice(0, 7);
    if (meses[k]) meses[k].gastos += (parseFloat(g.monto) || 0);
  });
  var keys = Object.keys(meses);
  drawSpark('sparkNeta', keys.map(function(k) { return meses[k].bruta - meses[k].gastos; }));
  drawSpark('sparkTicket', keys.map(function(k) { return meses[k].ops > 0 ? meses[k].fact / meses[k].ops : 0; }));
  drawSpark('sparkR', keys.map(function(k) { return meses[k].repEnt; }));
}

function drawSpark(id, vals) {
  var svg = el(id); if (!svg) return;
  var n = vals.length; if (!n) { svg.innerHTML = ''; return; }
  var min = Math.min.apply(null, vals), max = Math.max.apply(null, vals);
  var range = (max - min) || 1;
  var W = 100, H = 24, padX = 3, padY = 3;
  var pts = vals.map(function(v, i) {
    var x = padX + (i / (n - 1)) * (W - padX * 2);
    var y = H - padY - ((v - min) / range) * (H - padY * 2);
    return x.toFixed(1) + ',' + y.toFixed(1);
  });
  var last = pts[pts.length - 1].split(',');
  svg.innerHTML = '<polyline points="' + pts.join(' ') + '" fill="none" stroke="var(--muted)" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/>' +
    '<circle cx="' + last[0] + '" cy="' + last[1] + '" r="3" fill="var(--accent2)" stroke="var(--surface)" stroke-width="1.5"/>';
}

// Variación % vs. el período anterior de igual duración (para las 3 tarjetas KPI con tendencia)
function renderDelta(id, cur, prev) {
  var e = el(id); if (!e) return;
  if (!prev) { e.textContent = ''; e.className = 'kdelta'; return; }
  var pct = (cur - prev) / Math.abs(prev) * 100;
  var up = pct >= 0;
  e.textContent = (up ? '↑ ' : '↓ ') + Math.abs(pct).toFixed(0) + '%';
  e.className = 'kdelta ' + (up ? 'up' : 'down');
}

// Gráfico de evolución mensual — facturación + ganancia neta por mes (últimos 12 meses)
function renStatsMensual() {
  if (!el('barM')) return;
  var meses = {};
  var hoy = new Date();
  for (var i = 11; i >= 0; i--) {
    var dM = new Date(hoy.getFullYear(), hoy.getMonth() - i, 1);
    var k = dM.getFullYear() + '-' + ('0' + (dM.getMonth() + 1)).slice(-2);
    var lbl = dM.toLocaleDateString('es-AR', { month: 'short' }).replace('.', '');
    if (dM.getMonth() === 0) lbl += " '" + String(dM.getFullYear()).slice(-2);
    meses[k] = { fact: 0, bruta: 0, gastos: 0, label: lbl };
  }
  state.ventas.forEach(function(v) {
    var k = String(v.fecha || '').slice(0, 7);
    if (!meses[k]) return;
    var t = (parseFloat(v.precioVenta) || 0) * (parseInt(v.cantidad) || 0);
    var c = (parseFloat(v.costoUnit) || 0) * (parseInt(v.cantidad) || 0);
    meses[k].fact += t; meses[k].bruta += (t - c);
  });
  state.reparaciones.forEach(function(r) {
    if (r.estado !== 'Entregado') return;
    var k = String(r.fechaEntrega || r.fecha || '').slice(0, 7);
    if (!meses[k]) return;
    var t = parseFloat(r.precioReparacion) || 0, c = parseFloat(r.costoRepuesto) || 0;
    meses[k].fact += t; meses[k].bruta += (t - c);
  });
  (state.gastos || []).forEach(function(g) {
    if (g.categoria === 'Mercadería/Stock') return;
    var k = String(g.fecha || '').slice(0, 7);
    if (meses[k]) meses[k].gastos += (parseFloat(g.monto) || 0);
  });
  var vals = Object.keys(meses).map(function(k) { return meses[k]; });
  var maxM = Math.max.apply(null, vals.map(function(m) { return m.fact; }).concat([1]));
  el('barM').innerHTML = vals.map(function(m) {
    var neta = m.bruta - m.gastos;
    var hF = Math.max(m.fact / maxM * 100, 0);
    var hN = Math.max(neta / maxM * 100, 0);
    return '<div class="bcol">' +
      (m.fact > 0 ? '<div class="bval">' + fm(m.fact).replace('$', '') + '</div>' : '<div class="bval"></div>') +
      '<div style="width:100%;display:flex;align-items:flex-end;justify-content:center;gap:3px;height:100%">' +
        '<div title="Facturación ' + fm(m.fact) + '" style="width:42%;height:' + hF + '%;background:var(--accent);border-radius:3px 3px 0 0;' + (m.fact > 0 ? 'min-height:3px' : '') + '"></div>' +
        '<div title="Ganancia neta ' + fm(neta) + '" style="width:42%;height:' + hN + '%;background:var(--success);border-radius:3px 3px 0 0;' + (neta > 0 ? 'min-height:3px' : '') + '"></div>' +
      '</div>' +
      '<div class="blabel">' + m.label + '</div></div>';
  }).join('');
}

function exportXL() {
  var desde = new Date(el('dDesde').value + 'T00:00:00');
  var hasta = new Date(el('dHasta').value + 'T23:59:59');
  function inR(f) { var d = new Date(f + 'T12:00:00'); return d >= desde && d <= hasta; }
  var vF = state.ventas.filter(function(v) { return inR(v.fecha); });
  var rF = state.reparaciones.filter(function(r) { return inR(r.fecha); });
  var wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(
    [['Fecha', 'Hora', 'Producto', 'Cantidad', 'Precio Venta', 'Costo Unit.', 'Ganancia', 'Medio']].concat(
      vF.map(function(v) { return [v.fecha, v.hora || '', v.nombre, v.cantidad, parseFloat(v.precioVenta), parseFloat(v.costoUnit) || 0, (parseFloat(v.precioVenta) - parseFloat(v.costoUnit || 0)) * (parseInt(v.cantidad) || 0), v.medio]; })
    )), 'Ventas');
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(
    [['Fecha', 'Cliente', 'Equipo', 'Problema', 'Precio', 'Costo Repuesto', 'Ganancia', 'Recibió', 'Estado']].concat(
      rF.map(function(r) { return [r.fecha, r.cliente, r.equipo, r.problema, parseFloat(r.precioReparacion), parseFloat(r.costoRepuesto || 0), parseFloat(r.precioReparacion) - parseFloat(r.costoRepuesto || 0), r.tecnico, r.estado]; })
    )), 'Reparaciones');
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(
    [['Producto', 'Stock', 'Costo Unit.', 'Valor Total Costo']].concat(
      state.stock.map(function(s) { return [s.nombre, parseInt(s.cantidad), parseFloat(s.costo), (parseFloat(s.costo) || 0) * (parseInt(s.cantidad) || 0)]; })
    )), 'Stock');
  XLSX.writeFile(wb, 'Tecnopuerto_' + el('dDesde').value + '_' + el('dHasta').value + '.xlsx');
}
